;
; B-17 System Module Version 3 (16K)
; ----------------------------------
;
; This program allows a 500 baud program to be loaded into a buffer,
; and then output as a 1700 baud program, with a 500-baud preloader
; which automatically loads the payload program.
; 
; There are versions of the preloader which are resident at $4300 and $FD00
;
; First block is as follows:
;
; 4300 - 43DB: Preloader block (as described in the Preloader disassembly)

; Write a block of data to cassette
;
;  Expectations: 500-baud header already written, 1700 baud leader already written
;
; HL = starting point
; DE = size of block
;
46ce: 4e           ld   c,(hl)          ; get a byte
46cf: cd df 46     call $46DF           ; write the byte to cassette in B-17
46d2: cd 1b 47     call $471B           ; show the moving data marquee
46d5: cd 52 47     call $4752           ; calculate checksum
46d8: 23           inc  hl              ; next byte pointer inc
46d9: 1b           dec  de              ; decrement the counter
46da: 7a           ld   a,d
46db: b3           or   e
46dc: 20 f0        jr   nz,$46CE        ; repeat as necessary

;
; Write a byte - byte comes in 'C' register
;
; write 9 bite - a '1' start bit, with an 8-bit pulse train (present = '1'; absent = '0')
; following it, with precise timing
;
; ** NOTE ** - the timings within this section are different than in the 16K version !!
;   Also, a pulse starts with $06 then $05 then $04, as opposed to 16K's $05 then $06 then $04
;   And this subroutine takes one more byte than the 16K version
;
46df: f5           push af
46e0: c5           push bc
46e1: d5           push de
46e2: 16 09        ld   d,$09           ; 9 bits
46e4: 37           scf                  ; set carry flag - start bit is always '1'

; no pulse - delay instead
;

46e7: 06 17        ld   b,$17
46e9: 10 fe        djnz $46E9

46ec: 00           nop
46ed: 18 14        jr   $4703

; output a pulse

46fd: 00           nop                  ; delay
46fe: 00           nop

4703: 3e 04        ld   a,$04           ; motor on, no pulse (neutral)
